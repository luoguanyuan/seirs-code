function dSEIRdt = seirs_ode(t, y, alpha, beta, gamma, mu, epsilon, eta, delta)
    S = y(1);
    E = y(2);
    I = y(3);
    R = y(4);

    dSdt = -alpha * S*I + eta * E + epsilon * I + mu * R;
    dEdt = alpha * S*I - beta * E - eta * E - delta * E;
    dIdt = beta * E - epsilon * I - gamma * I;
    dRdt = delta * E + gamma * I - mu * R;

    dSEIRdt = [dSdt; dEdt; dIdt; dRdt];
end
